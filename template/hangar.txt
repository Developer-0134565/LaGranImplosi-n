window.GameNumber = (function () {
    'use strict';

    var ladder = [[1.0e+33,"Dc"],[1.0e+30,"No"],[1.0e+27,"Oc"],[1.0e+24,"Sp"],[1.0e+21,"Sx"],[1.0e+18,"Qi"],[1000000000000000,"Qa"],[1000000000000,"Tn"],[1000000000,"Bn"],[1000000,"Mn"]];

    function groupDigits(str, sep) {
        sep = sep || ',';
        return String(str).replace(/\B(?=(\d{3})+(?!\d))/g, sep);
    }

    // Full digit expansion of a finite double: exact to 2^53, then 15 significant
    // digits zero-padded (mirrors AppUtil::formatNumberFull's %.14e algorithm).
    function full(n) {
        n = Number(n);
        if (isNaN(n)) { return '0'; }
        if (!isFinite(n)) { return n < 0 ? '-∞' : '∞'; }
        var neg = n < 0;
        n = Math.abs(n);
        if (n < 9007199254740992) {
            // round, not floor — PHP number_format rounds; fractional Tier B
            // survivors must not flicker between server render and JS repaint.
            return (neg ? '-' : '') + groupDigits(Math.round(n).toString());
        }
        var parts = n.toExponential(14).split('e');
        var digits = parts[0].replace('.', '');
        var intLen = parseInt(parts[1], 10) + 1;
        if (intLen > digits.length) {
            digits += new Array(intLen - digits.length + 1).join('0');
        } else {
            digits = digits.substring(0, Math.max(1, intLen));
        }
        return (neg ? '-' : '') + groupDigits(digits);
    }

    // Compact form (mirrors AppUtil::formatNumberShort).
    function short(n) {
        n = Number(n);
        if (isNaN(n)) { return '0'; }
        if (!isFinite(n)) { return n < 0 ? '-∞' : '∞'; }
        if (n < 0) { return '-' + short(-n); }
        if (n >= 1e36) { return n.toExponential(2); }
        if (n >= 1e6) {
            for (var i = 0; i < ladder.length; i++) {
                if (n >= ladder[i][0]) {
                    var scaled = n / ladder[i][0];
                    var decimals = (scaled < 100 && Math.floor(scaled) !== scaled) ? 1 : 0;
                    // toFixed rounds — mirrors PHP number_format($scaled, $decimals).
                    var shown = scaled.toFixed(decimals);
                    return groupDigits(shown.split('.')[0]) + (decimals === 1 ? '.' + shown.split('.')[1] : '') + ladder[i][1];
                }
            }
        }
        if (n > 10000) { return groupDigits(String(Math.floor(n / 1000))) + 'K'; }
        if (n > 1000) {
            var thousands = Math.floor((n / 1000) * 10) / 10;
            return (n % 1000 === 0 ? String(Math.floor(thousands)) : thousands.toFixed(1)) + 'K';
        }
        return groupDigits(String(Math.round(n))); // number_format rounds
    }

    return { ladder: ladder, short: short, full: full };
})();

    <script type="text/javascript">
window.AmountUnits = (function () {
    'use strict';

    // The ladder is AppUtil::suffixLadder() (Dc=1e33 first, Mn=1e6 last), i.e. the
    // same vocabulary every formatted number on the page already uses. K is not on
    // that ladder — formatNumberShort renders thousands as "12K" without one — so it
    // is added here, because "10k" is the shorthand people reach for first.
    var CAP_TEXT = "M\u00e1ximo aqu\u00ed: :amount";
    var ladder = (window.GameNumber && window.GameNumber.ladder) || [];
    var units = [{ suffix: 'K', exp: 3 }];
    for (var i = ladder.length - 1; i >= 0; i--) {
        units.push({ suffix: ladder[i][1], exp: Math.round(Math.log(ladder[i][0]) / Math.LN10) });
    }

    // alias -> exponent. Full suffix always; the initial letter only where it names
    // exactly one unit (so `q` and `s` stay incomplete and wait for the second char).
    var byAlias = {};
    var firstLetterCount = {};
    units.forEach(function (u) {
        var lower = u.suffix.toLowerCase();
        byAlias[lower] = u.exp;
        firstLetterCount[lower.charAt(0)] = (firstLetterCount[lower.charAt(0)] || 0) + 1;
    });
    units.forEach(function (u) {
        var head = u.suffix.toLowerCase().charAt(0);
        if (firstLetterCount[head] === 1) { byAlias[head] = u.exp; }
    });

    // Could another character still extend these letters into a different unit? `n` is
    // No but also the head of nothing else, `m` is Mn, `q` is both Qa and Qi — and a
    // letter that CAN be extended must never expand on its own, or `33.091no` expands
    // at the `n` and then again at the `o`, i.e. 1e27 times too big.
    function extendable(letters) {
        for (var alias in byAlias) {
            if (alias.length > letters.length && alias.indexOf(letters) === 0) { return true; }
        }
        return false;
    }

    function groupDigits(digits) {
        return String(digits).replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    }

    // Multiply a decimal STRING by 10^exp, exactly, at any magnitude: shift the point.
    // A fraction finer than the unit is TRUNCATED (never rounded up) — "1.2345K" loses
    // the trailing 5, and an amount that grows on its own is the one thing a box must
    // not do. ⚠️ No comma-grouped figure in this comment: it ships inside the in-game
    // LAYOUT, so any number here is on every page and trips a full-page assertDontSee
    // in an unrelated test (it broke WikiExpeditionDocsTest's relic-odds guard).
    function scale(text, exp) {
        var cleaned = String(text).replace(/[^0-9.]/g, '');
        var parts = cleaned.split('.');
        var whole = (parts[0] || '').replace(/[^0-9]/g, '');
        var frac = (parts[1] || '').replace(/[^0-9]/g, '');
        var digits = (whole + frac).replace(/^0+(?=\d)/, '');
        if (digits === '') { return '0'; }
        var shift = exp - frac.length;
        if (shift >= 0) {
            digits += new Array(shift + 1).join('0');
        } else {
            digits = digits.slice(0, Math.max(0, digits.length + shift));
        }
        digits = digits.replace(/^0+(?=\d)/, '');
        return digits === '' ? '0' : digits;
    }

    // Split a box's text into [number, unit letters]. The box may already be comma
    // grouped, so separators are dropped first; a bare decimal point is kept because
    // "2.5Qa" is the whole point of this feature.
    function split(value) {
        var text = String(value === null || value === undefined ? '' : value).replace(/[\s,]/g, '');
        var m = /^(\d*(?:\.\d*)?)([a-zA-Z]*)$/.exec(text);
        if (!m || m[1] === '' || m[1] === '.') { return null; }
        return { number: m[1], letters: m[2].toLowerCase() };
    }

    // Is this text a shorthand still being typed? True in the window where the page's
    // own re-formatters must be kept away from the box.
    //
    // TWO shapes qualify, and the DECIMAL POINT is the one that is easy to miss: every
    // amount box here strips non-digits on keyup, so `2.5qi` lost its point at the
    // second keystroke and arrived as 25qi — ten times what was asked for. A single
    // point is therefore held too, until a unit resolves it or focus leaves.
    //
    // A SECOND point means the player is using dots as group separators (1.234.567),
    // which is not shorthand at all: split() rejects it, so it falls straight through
    // to the page's own stripper exactly as it always did.
    function isPending(value) {
        var s = split(value);
        if (!s) { return false; }
        if (s.letters !== '') { return extendable(s.letters); }
        return s.number.indexOf('.') !== -1;
    }

    // "10qa" -> { digits: "10000000000000000000", mantissa: "10", exp: 18 }, or null when
    // there is no unit to apply. The PARTS matter as much as the result: typing a suffix
    // has to seed the same memory a chip click does, or the chips cannot switch the unit
    // of a number that was typed rather than tapped.
    function expandParts(value) {
        var s = split(value);
        if (!s || s.letters === '' || byAlias[s.letters] === undefined) { return null; }
        var exp = byAlias[s.letters];
        return { digits: scale(s.number, exp), mantissa: s.number, exp: exp };
    }

    function expand(value) {
        var parts = expandParts(value);
        return parts === null ? null : parts.digits;
    }

    // What this box should hold if the player stopped typing right now: the shorthand
    // if it names a unit, otherwise its bare digits — which is exactly what the page's
    // own stripper would have produced. Leaving a box is never worse than never having
    // typed a suffix at all.
    function resolved(value) {
        var digits = expand(value);
        if (digits !== null) { return digits; }
        digits = String(value).replace(/[^0-9]/g, '').replace(/^0+(?=\d)/, '');
        return digits === '' ? '0' : digits;
    }

    // The page's own handlers are what clamp, re-render and recompute; they listen on
    // keyup/change/input, so a programmatic write announces itself the same way a
    // keystroke would. Re-entry is guarded: our own synthetic events must not be read
    // back as fresh typing.
    var announcing = false;
    function announce(input) {
        if (announcing) { return; }
        announcing = true;
        try {
            ['input', 'keyup', 'change'].forEach(function (type) {
                input.dispatchEvent(new Event(type, { bubbles: true }));
            });
        } finally {
            announcing = false;
        }
    }

    // What the strip last put in this box, and the mantissa it was built from.
    //
    // WHY THE MANTISSA IS REMEMBERED: without it a chip can only multiply what it finds,
    // so `234` + Qi + Qa read the already-expanded 234 Qi and multiplied it AGAIN — there
    // was no way back to `234 Qa` short of retyping, and clicking the same chip twice was
    // the same bug. A chip is a CHOICE OF UNIT for a number you typed, so the number is
    // what gets remembered and the chips just re-apply it.
    //
    // It is cleared by real typing and only by real typing (see the capture handler): a
    // page clamping our write happens inside `announce()` with `announcing` set, so a
    // reduced amount never destroys the mantissa the player actually asked for.
    var applied = null;

    // The box exactly as it stood when this strip last wrote it — INCLUDING whatever the
    // page then clamped it to. A keystroke fires `input` AND `keyup`, so our own expansion
    // (done on `input`) is followed by a `keyup` carrying no new text at all: without this,
    // that second event was read as the player typing and it wiped both the chosen unit and
    // the "Max here" note a few microseconds after they appeared.
    var lastWrite = null;

    // Compare two digit strings without going through a double — these routinely exceed
    // 2^53, where `Number()` cannot tell 1e21 from 1e21 + 1.
    function digitsLess(a, b) {
        a = String(a).replace(/^0+(?=\d)/, '');
        b = String(b).replace(/^0+(?=\d)/, '');
        if (a.length !== b.length) { return a.length < b.length; }
        return a < b;
    }

    function write(input, digits, mantissa, exp) {
        input.value = groupDigits(digits);
        try { input.setSelectionRange(input.value.length, input.value.length); } catch (e) { /* older browsers */ }
        applied = mantissa === undefined ? null : { input: input, mantissa: mantissa, exp: exp };
        announce(input);
        lastWrite = { input: input, value: input.value };
        // The page has had its say by now — its own handlers run synchronously off the
        // events above — so anything smaller than what went in was REDUCED by a ceiling
        // this strip knows nothing about (stock on a trade page, cargo space on fleet 3).
        // Silence there is the complaint: "I typed 20tn and got 18tn and it said nothing."
        var landed = String(input.value).replace(/[^0-9]/g, '');
        if (digitsLess(landed, digits)) { showCap(landed); } else { showCap(null); }
        markActive();
    }

    function isTarget(el) {
        return !!(el && el.tagName === 'INPUT' && el.hasAttribute('data-amount-units') && !el.disabled && !el.readOnly);
    }

    // ---- the chip strip -----------------------------------------------------
    // One shared popover, anchored under whichever box has focus. Per-row chips would
    // be four rows of eleven buttons on the fleet page; this is the same control in
    // the one place the player is already looking.
    var pop = null;
    var popFor = null;

    // How far up the ladder is worth offering. A box that knows its own ceiling
    // (data-max, or max) shows one unit past it: chips for magnitudes the player
    // cannot reach are noise, and on a young account the strip is just "K Mn Bn".
    function unitsFor(input) {
        var raw = input.getAttribute('data-amount-units-max') || input.getAttribute('data-max') || input.getAttribute('max') || '';
        var digits = String(raw).replace(/[^0-9]/g, '').replace(/^0+(?=\d)/, '');
        if (digits === '' || digits === '0') { return units; }
        var out = units.filter(function (u) { return u.exp < digits.length + 2; });
        return out.length ? out : units.slice(0, 3);
    }

    function ensurePop() {
        if (pop) { return pop; }
        pop = document.createElement('div');
        pop.className = 'amt-units';
        // Body-level, so a transformed ancestor cannot scope its fixed position
        // (rules.md — the same reason in-chrome modals are reparented).
        document.body.appendChild(pop);
        // mousedown, not click: the box must not lose focus before the chip acts.
        pop.addEventListener('mousedown', function (e) {
            // Every press inside the strip keeps focus in the box — including the gaps
            // between chips, which otherwise blurred the input and closed the strip on a
            // click that did nothing.
            e.preventDefault();
            var chip = e.target.closest ? e.target.closest('[data-exp]') : null;
            if (!chip || !popFor) { return; }
            var exp = parseInt(chip.getAttribute('data-exp'), 10);
            var number;
            if (applied && applied.input === popFor) {
                // A unit is already applied, so this chip CHANGES it rather than stacking
                // on top of it: 234 Qi -> Qa is 234 Qa, and the same chip twice is a no-op.
                number = applied.mantissa;
            } else {
                var s = split(popFor.value);
                // An empty box means "1 of these" — the chip is then a shortcut to the
                // round number, which is what it reads like.
                number = (s && s.number !== '' && s.number !== '.') ? s.number : '1';
            }
            write(popFor, scale(number, exp), number, exp);
        });
        return pop;
    }

    // The chosen unit has to be VISIBLE, or "which one am I on" is a guess — and with the
    // mantissa remembered, the strip is a radio group, not a row of multiply buttons.
    function markActive() {
        if (!pop) { return; }
        var on = applied && applied.input === popFor ? String(applied.exp) : null;
        Array.prototype.forEach.call(pop.querySelectorAll('[data-exp]'), function (chip) {
            chip.classList.toggle('on', chip.getAttribute('data-exp') === on);
        });
    }

    // "Max here: 18Tn" — said in the strip, which is already under the box the player is
    // looking at, and worded without naming a CAUSE: on a trade page the ceiling is the
    // stock, on fleet page 3 it is usually cargo space, and one sentence cannot claim both.
    var capNote = null;
    function showCap(landedDigits) {
        if (!pop) { return; }
        if (!landedDigits) {
            if (capNote) { capNote.remove(); capNote = null; place(); }
            return;
        }
        if (!capNote) {
            capNote = document.createElement('span');
            capNote.className = 'amt-cap';
            pop.appendChild(capNote);
        }
        capNote.textContent = CAP_TEXT.replace(':amount', shortDigits(landedDigits));
        place();
    }

    // Compact form of a DIGIT STRING. GameNumber.short() takes a double, which past 2^53
    // is already the wrong number — here the digits are exact, so the magnitude is read
    // off the LENGTH and the leading digits are kept as characters.
    function shortDigits(digits) {
        digits = String(digits).replace(/^0+(?=\d)/, '');
        var best = null;
        units.forEach(function (u) {
            if (digits.length > u.exp) { best = u; }
        });
        if (!best) { return groupDigits(digits); }
        var whole = digits.slice(0, digits.length - best.exp);
        var frac = digits.slice(digits.length - best.exp, digits.length - best.exp + 1);
        return groupDigits(whole) + (whole.length < 3 && frac !== '0' ? '.' + frac : '') + best.suffix;
    }

    // The strip is floated, but it RESERVES its own space: while it is open the anchor
    // carries a bottom margin the height of the strip, so the rows under it move down
    // instead of being covered. Without that, the strip on the metal box sits exactly
    // over the crystal box, and the click that means "now fill crystal" lands on a chip
    // and multiplies metal instead — a wrong amount from a gesture that looked right.
    // The room is made on the input's ROW, not the input: these boxes sit in flex pairs
    // with a Fill/MAX button beside them, and a margin on the input alone stretches that
    // button under the strip instead of moving it. The row is the outermost ancestor
    // that still ends where the input ends.
    function displayOf(el) {
        return (window.getComputedStyle ? window.getComputedStyle(el).display : '') || '';
    }

    function rowFor(input) {
        var el = input, p = input.parentElement, bottom = input.getBoundingClientRect().bottom, hops = 0;
        while (p && p !== document.body && hops++ < 4) {
            var d = displayOf(p);
            // A table row or the table itself cannot hold room open — margin and padding
            // are both ignored on them. The CELL can (see reserve), so climbing stops there.
            if (d === 'table-row' || d === 'table-row-group' || d === 'table') { break; }
            // ⚠️ …and a cell is taken whether or not it ends where the input does. A cell
            // is usually taller (its own padding, plus the row's height), so the "stop when
            // the parent is taller" rule below would leave the margin on the INPUT — which
            // `vertical-align: middle` then lifts out of line with its neighbours while the
            // strip stays put over them.
            if (d !== 'table-cell' && p.getBoundingClientRect().bottom - bottom > 6) { break; }
            el = p;
            p = p.parentElement;
            if (d === 'table-cell') { break; }
        }
        return el;
    }

    var spacerOn = null;
    var spacerProp = 'marginBottom';
    var spacerHeld = [];
    var spacerPx = 0;
    function reserve(el, px) {
        spacerHeld.forEach(function (h) { h.el.style[spacerProp] = h.was; });
        spacerHeld = [];
        spacerOn = null;
        spacerPx = 0;
        if (!el || px === null) { return; }
        // ⚠️ A table cell IGNORES margin — the marketplace's give/want boxes live in a
        // <table>, and a margin there reserved nothing at all, leaving the strip over the
        // row below. Padding grows a cell (and with it the row) the way margin grows a div.
        var cell = displayOf(el) === 'table-cell';
        spacerProp = cell ? 'paddingBottom' : 'marginBottom';
        spacerOn = el;
        spacerPx = px;
        // ⚠️ …and it has to grow EVERY cell in the row, not just this one: a taller row
        // re-centres its siblings (default `vertical-align: middle`), so padding one cell
        // drops the neighbouring inputs half a strip and the strip lands on them.
        var targets = cell && el.parentElement ? Array.prototype.slice.call(el.parentElement.children) : [el];
        targets.forEach(function (t) {
            if (cell && displayOf(t) !== 'table-cell') { return; }
            var existing = cell ? (parseFloat(window.getComputedStyle(t).paddingBottom) || 0) : 0;
            spacerHeld.push({ el: t, was: t.style[spacerProp] });
            t.style[spacerProp] = (px + existing) + 'px';
        });
    }

    // Decided once per open, BEFORE any room is reserved, and stable: reserving pushes the
    // content under the box down without moving the box, so it cannot change the answer.
    // A strip that goes above must NOT reserve — the gap under the box would be held open
    // for something sitting somewhere else.
    var below = true;

    function place() {
        if (!pop || !popFor || pop.style.display === 'none') { return; }
        var r = popFor.getBoundingClientRect();
        // margin-bottom sits OUTSIDE the border box, so the row's rect is unchanged by the
        // room it holds open — but PADDING is inside it, so a cell's rect already contains
        // the gap and the strip would be placed one gap too low, i.e. back on top of the
        // row below. Either way this is the row's real bottom edge.
        var rowBottom = (spacerOn || popFor).getBoundingClientRect().bottom
            - (spacerOn && spacerProp === 'paddingBottom' ? spacerPx : 0);
        pop.style.left = Math.max(4, Math.min(r.left, window.innerWidth - pop.offsetWidth - 4)) + 'px';

(function () {
    'use strict';

    function wire(btn, backdrop) {
        if (!btn || !backdrop || btn.__eventBound) { return; }
        btn.__eventBound = true;

        if (backdrop.parentNode !== document.body) {
            document.body.appendChild(backdrop);
        }

        function open() { backdrop.style.display = 'flex'; }
        function close() { backdrop.style.display = 'none'; }

        btn.addEventListener('click', function (e) { e.preventDefault(); open(); });
        // Chips are not always anchors — keep the keyboard activation an <a> gets free.
        btn.addEventListener('keydown', function (e) {
            if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); open(); }
        });
        backdrop.addEventListener('click', function (e) {
            if (e.target === backdrop) { close(); }
        });
        var closeBtn = backdrop.querySelector('.event-modal-close');
        if (closeBtn) { closeBtn.addEventListener('click', close); }
        document.addEventListener('keydown', function (e) {
            if (e.key === 'Escape') { close(); }
        });
    }

    function boot() {
        document.querySelectorAll('[data-event-open]').forEach(function (btn) {
            wire(btn, document.getElementById(btn.getAttribute('data-event-open')));
        });
        // Legacy pairing (old layout's pill).
        wire(document.getElementById('eventButton'), document.getElementById('eventModalBackdrop'));
    }

    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', boot);
    } else {
        boot();
    }
})();

<script type="text/javascript">
(function () {
    'use strict';
    var S = {"closes":"Cierra en __T__","closed":"Cerrada el __T__ (tu hora local)","error":"Tu voto no se ha registrado. Int\u00e9ntalo de nuevo.","saved":"Voto guardado."};

    function fmtDuration(seconds) {
        if (typeof window.formatTimeWrapper === 'function') {
            return window.formatTimeWrapper(Math.max(0, seconds), 2, true, ' ', false, '');
        }
        var h = Math.floor(seconds / 3600), m = Math.floor((seconds % 3600) / 60);
        return h + 'h ' + m + 'm';
    }

    // "Closes in 2d 4h" / "Closed <local date>": the server ships timestamps, the viewer's
    // browser decides what they read as.
    function fillTimes(root) {
        (root || document).querySelectorAll('[data-lts-poll]').forEach(function (el) {
            var ts = parseInt(el.getAttribute('data-lts-poll'), 10);
            if (!ts) { return; }
            if (el.getAttribute('data-poll-when') === 'closes') {
                var left = ts - Math.floor(Date.now() / 1000);
                el.textContent = S.closes.replace('__T__', fmtDuration(left));
            } else {
                var d = new Date(ts * 1000);
                el.textContent = S.closed.replace('__T__', d.toLocaleDateString() + ' ' + d.toLocaleTimeString());
            }
        });
    }

    function syncButton(card) {
        var btn = card.querySelector('.poll-vote-btn');
        if (!btn) { return; }
        var checked = card.querySelector('input[name="option"]:checked');
        var mine = card.getAttribute('data-my-vote') || '';
        btn.disabled = !checked || checked.value === mine;
    }

    // The sidebar chip for this poll: stop the pulse and, after a vote, say VOTED (both labels
    // ride on the chip as data attributes so this script needs no lang strings of its own). A
    // poll that closed under the page only loses the pulse — the chip is gone on the next load.
    function settleChip(pollId, voted) {
        var chip = document.getElementById('neoPollChip-' + pollId);
        if (chip) {
            chip.classList.remove('needs-vote');
            if (voted) {
                var tag = chip.querySelector('.tag');
                if (tag && chip.getAttribute('data-tag-voted')) { tag.textContent = chip.getAttribute('data-tag-voted'); }
                if (chip.getAttribute('data-title-voted')) { chip.setAttribute('title', chip.getAttribute('data-title-voted')); }
            }
        }
        if (!document.querySelector('.event-live.poll.needs-vote')) {
            document.querySelectorAll('.neo-utilnav a.poll-open').forEach(function (a) {
                a.classList.remove('poll-open');
                var dot = a.querySelector('.unread-dot');
                if (dot) { dot.parentNode.removeChild(dot); }
            });
        }
    }

    // Swap every card of this poll on the page (popup + /polls page can both be open) for the
    // server's re-render. Returns the new nodes; an empty or malformed payload swaps nothing.
    function replaceCards(pollId, html) {
        var tpl = document.createElement('div');
        tpl.innerHTML = html || '';
        var fresh = tpl.querySelector('.poll-card');
        if (!fresh) { return []; }
        var swapped = [];
        document.querySelectorAll('.poll-card[data-poll-id="' + pollId + '"]').forEach(function (old, i) {
            var node = i === 0 ? fresh : fresh.cloneNode(true);
            old.parentNode.replaceChild(node, old);
            fillTimes(node);
            swapped.push(node);
        });
        return swapped;
    }

    document.addEventListener('change', function (e) {
        if (!(e.target instanceof HTMLInputElement) || e.target.name !== 'option') { return; }
        var card = e.target.closest('.poll-card');
        if (card) { syncButton(card); }
    });

    document.addEventListener('submit', function (e) {
        var form = e.target;
        if (!(form instanceof HTMLFormElement) || !form.classList.contains('poll-form')) { return; }
        e.preventDefault();
        var card = form.closest('.poll-card');
        var btn = form.querySelector('.poll-vote-btn');
        var checked = form.querySelector('input[name="option"]:checked');
        var msg = form.querySelector('.poll-msg');
        if (!card || !checked || (btn && btn.disabled)) { return; }
        if (btn) { btn.disabled = true; }
        if (msg) { msg.textContent = ''; msg.classList.remove('ok'); }

        var pollId = card.getAttribute('data-poll-id');

        function note(nodes, text, ok) {
            nodes.forEach(function (node) {
                var m = node.querySelector('.poll-msg');
                if (!m) { return; }
                m.textContent = text;
                m.classList.toggle('ok', !!ok);
            });
        }

        $.post(form.getAttribute('action'), { option: checked.value, _token: window.token }, function (res) {
            try {
                if (res && res.newAjaxToken) { window.token = res.newAjaxToken; }
                if (!res || !res.success) {
                    var text = (res && res.message) ? res.message : S.error;
                    // A poll that closed under this page comes back as its CLOSED card: show it,
                    // stop the pulse, and stop offering a vote that will keep being refused.
                    var swappedClosed = (res && res.reason === 'closed' && res.html) ? replaceCards(pollId, res.html) : [];
                    if (swappedClosed.length) {
                        settleChip(pollId, false);
                        note(swappedClosed, text, false);
                        return;
                    }
                    if (msg) { msg.textContent = text; }
                    if (btn && !(res && res.reason === 'closed')) { syncButton(card); }
                    return;
                }
                var swapped = replaceCards(pollId, res.html);
                if (!swapped.length) {
                    // Nothing to swap in: the vote IS saved, so say so on the old card.
                    if (msg) { msg.textContent = S.saved; msg.classList.add('ok'); }
                    card.setAttribute('data-my-vote', checked.value);
                    syncButton(card);
                } else {
                    note(swapped, S.saved, true);
                }
                settleChip(pollId, true);
            } catch (e) {
                if (msg) { msg.textContent = S.error; }
                syncButton(card);
            }

<script type="text/javascript">
    // BBCode parser for buddy request previews, alliance internal/external/application text
    // and the markItUp editor's client-side Preview button.
    //
    // SECURITY: this parser is the escaping boundary for every caller, and it must stay that way.
    // Its output is assigned to .innerHTML, and its callers hand it fully untrusted text:
    //  - alliance/info + alliance/overview read .textContent / .text() off a server-escaped
    //    element, which DECODES the entities back to raw — the server's Blade escaping is
    //    already undone by the time the string arrives here.
    //    (Do NOT write a literal double-brace echo tag in this comment: Blade compiles echo
    //    tags inside JS comments too, and an empty one becomes e() with no args = 500.)
    //  - the highscore + galaxy previews pass raw $textarea.val() straight from the keyboard.
    // So escaping at any call site would be redundant there and missing here. Escape first,
    // then substitute tags: the tags below are then the only markup in the result.
    window.buddyBBCodeParser = function(text) {
        if (!text) return '';

        var escapeHtml = function(value) {
            return String(value)
                .replace(/&/g, '&amp;')
                .replace(/</g, '&lt;')
                .replace(/>/g, '&gt;')
                .replace(/"/g, '&quot;')
                .replace(/'/g, '&#39;');
        };

        // Only http(s) may reach an href. Escaping alone does not stop "javascript:alert(1)" —
        // it contains no characters escapeHtml touches. Anything else renders as inert text.
        var safeUrl = function(url) {
            return /^https?:\/\//i.test(url) ? url : '';
        };

        // Colours land inside a style attribute. Escaping kills the quote-breakout
        // ([color=red" onmouseover="alert(1)]), and this keeps the value from carrying extra
        // CSS declarations via ";".
        var safeColor = function(color) {
            return /^(#[0-9a-f]{3,8}|[a-z]+)$/i.test(color) ? color : '';
        };

        // mailto: hrefs get a strict shape check — same reasoning as safeUrl.
        var safeEmail = function(value) {
            return /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(value) ? value : '';
        };

        var escaped = escapeHtml(text);

        // [code] contents must render literally — stash them before tag substitution so
        // bbcode inside a code block is not parsed, restore after the newline pass.
        var codeBlocks = [];
        escaped = escaped.replace(/\[code\]([\s\S]*?)\[\/code\]/gi, function(match, body) {
            codeBlocks.push(body);
            return '\u0000C' + (codeBlocks.length - 1) + '\u0000';
        });

        // NOTE: bodies use [\s\S]*? not .*? — "." never crosses a newline, so a tag wrapping
        // two lines would silently stay literal.
        var html = escaped
            .replace(/\[b\]([\s\S]*?)\[\/b\]/gi, '<strong style="font-weight:bold">$1</strong>')
            .replace(/\[i\]([\s\S]*?)\[\/i\]/gi, '<em style="font-style:italic">$1</em>')
            .replace(/\[u\]([\s\S]*?)\[\/u\]/gi, '<span style="text-decoration:underline">$1</span>')
            .replace(/\[s\]([\s\S]*?)\[\/s\]/gi, '<span style="text-decoration:line-through">$1</span>')
            .replace(/\[sup\]([\s\S]*?)\[\/sup\]/gi, '<sup>$1</sup>')
            .replace(/\[sub\]([\s\S]*?)\[\/sub\]/gi, '<sub>$1</sub>')
            .replace(/\[color=([^\]]*)\]([\s\S]*?)\[\/color\]/gi, function(match, color, body) {
                var value = safeColor(color);
                return value ? '<span style="color:' + value + '">' + body + '</span>' : body;
            })
            .replace(/\[size=(\d+)\]([\s\S]*?)\[\/size\]/gi, function(match, size, body) {
                var px = Math.min(30, Math.max(6, parseInt(size, 10)));
                return '<span style="font-size:' + px + 'px">' + body + '</span>';
            })
            .replace(/\[url=([^\]]*)\]([\s\S]*?)\[\/url\]/gi, function(match, url, body) {
                var href = safeUrl(url);
                return href
                    ? '<a href="' + href + '" target="_blank" rel="noopener noreferrer" style="color:#6f9fc8;text-decoration:underline">' + body + '</a>'
                    : body;
            })
            .replace(/\[url\]([\s\S]*?)\[\/url\]/gi, function(match, url) {
                var href = safeUrl(url);
                return href
                    ? '<a href="' + href + '" target="_blank" rel="noopener noreferrer" style="color:#6f9fc8;text-decoration:underline">' + href + '</a>'
                    : url;
            })
            .replace(/\[email=([^\]]*)\]([\s\S]*?)\[\/email\]/gi, function(match, email, body) {
                var addr = safeEmail(email);
                return addr
                    ? '<a href="mailto:' + addr + '" style="color:#6f9fc8;text-decoration:underline">' + body + '</a>'
                    : body;
            })
            .replace(/\[email\]([\s\S]*?)\[\/email\]/gi, function(match, email) {
                var addr = safeEmail(email);
                return addr
                    ? '<a href="mailto:' + addr + '" style="color:#6f9fc8;text-decoration:underline">' + addr + '</a>'
                    : email;
            })
            .replace(/\[player\]([\s\S]*?)\[\/player\]/gi, '<span style="color:#a58cff;font-weight:bold">$1</span>')
            .replace(/\[coordinates\]\s*(\d{1,2}):(\d{1,3}):(\d{1,2})\s*\[\/coordinates\]/gi,
                '<a href="/galaxy?galaxy=$1&amp;system=$2&amp;position=$3" style="color:#6f9fc8;text-decoration:underline">[$1:$2:$3]</a>')
            .replace(/\[coordinates\]([\s\S]*?)\[\/coordinates\]/gi, '$1')
            .replace(/\[align=(left|center|right|justify)\]([\s\S]*?)\[\/align\]/gi, '<div style="text-align:$1">$2</div>')

    // ---- Planet-list construction indicators: retire them when the queue actually ends. ----
    // The build / ships / defense chips (#121) are server-rendered, so a queue that emptied
    // while the player sat on any page kept showing its icon until the next full page load.
    //
    // This is NOT a poll: the server hands back `next` — the soonest moment any of this
    // player's indicators could change — and we arm a single timer on it. One request per
    // real completion event, none at all while every queue is idle. The first `next` is
    // rendered with the page (neoPindNext), so an idle account costs zero requests.
    (function () {
        var timer = null;
        var inFlight = false;
        var rerun = false;
        var lastFetch = 0;
        // Floor between fetches: a 20-deep queue of instant (0s) builds would otherwise
        // fire back-to-back requests as each head completes.
        var MIN_GAP_MS = 4000;

        function paint(bodies) {
            Object.keys(bodies).forEach(function (id) {
                var state = bodies[id];

                // Every list that renders the cluster (sidebar + mobile strip) shares the id.
                document.querySelectorAll('.neo-pind[data-pind="' + id + '"]').forEach(function (el) {
                    el.innerHTML = state.html;
                });

                // The sidebar moon badge is too small for the full cluster and carries a
                // single build dot instead — same state, different markup.
                document.querySelectorAll('.neo-moonbadge[data-mbuild="' + id + '"]').forEach(function (badge) {
                    var dot = badge.querySelector('.mbuild');
                    if (!state.building) {
                        if (dot) { dot.remove(); }
                        return;
                    }
                    if (!dot) {
                        dot = document.createElement('span');
                        dot.className = 'mbuild';
                        badge.appendChild(dot);
                    }
                    dot.classList.toggle('down', !!state.downgrading);
                    dot.setAttribute('title', state.downgrading ? 'Deconstructing' : 'Under construction');
                });
            });
        }

        // `seconds` is how long the SERVER says it is until the next possible change, and is
        // deliberately not derived from a timestamp comparison here: the layout's `timeDiff`
        // is an hour out (Carbon::now() is emitted without a zone, so the browser parses UTC
        // as local), which silently pushed every timer an hour into the future.
        function arm(seconds) {
            if (timer) { clearTimeout(timer); timer = null; }
            if (seconds < 0) { return; }
            // +1s so the server has certainly processed the completion (queues advance in
            // the GlobalGame middleware of the very request we are about to make).
            var delay = (seconds + 1) * 1000;
            var sinceLast = Date.now() - lastFetch;
            if (delay < MIN_GAP_MS - sinceLast) { delay = MIN_GAP_MS - sinceLast; }
            if (delay < 250) { delay = 250; }
            timer = setTimeout(refresh, delay);
        }

        function refresh() {
            // A refresh asked for mid-flight is remembered, not dropped: the in-flight answer
            // predates whatever just changed (a cancel landing while a completion refresh is
            // open would otherwise leave the sidebar showing a queue that no longer exists).
            if (inFlight) { rerun = true; return; }
            inFlight = true;
            lastFetch = Date.now();
            fetch(neoStampCp('https://antares.deuteria.net/ajax/planet-indicators'), { headers: { 'X-Requested-With': 'XMLHttpRequest' } })
                .then(function (r) { return r.json(); })
                .then(function (d) {
                    paint(d.bodies || {});
                    arm(typeof d.next_in === 'number' ? d.next_in : -1);
                })
                .catch(function () {})
                .then(function () {
                    inFlight = false;
                    if (rerun) { rerun = false; setTimeout(refresh, 500); }
                });
        }

        // Queue actions on the current page (add / cancel / a countdown hitting zero) already
        // refresh that page's cards — piggyback so the sidebar agrees immediately instead of
        // waiting for its own timer.
        window.neoRefreshPlanetIndicators = function () {
            if (timer) { clearTimeout(timer); timer = null; }
            refresh();
        };

                arm(43400);
    })();

            function p(n) { return n < 10 ? '0' + n : n; }

            var d = document.createElement('div');

        // Comma-group an integer for display (ship counts, resource shipments, fleet-value

        // Fold near-identical EXPEDITIONS (only — owner call) into one row: same
        // leg/state/route, arrivals within 120s. Mirrors the movement page's fold.
        var openGroups = {};
        function groupEvents(list) {
            var out = [], heads = {};
            list.forEach(function (e) {
                if (e.mission_type !== 15) { out.push(e); return; }
                var key = [e.label, e.state, e.origin_coords, e.dest_coords].join('|');
                var h = heads[key];
                if (h && Math.abs(e.arrival - h.arrival) <= 120) { h._members.push(e); }
                else { e._members = []; heads[key] = e; out.push(e); }
            });
            // A folded row speaks for every wave it hides, so its chip and tooltip must
            // state the GROUP — one wave's count beside an ×12 label read as eleven
            // fleets gone missing. Sub-rows keep their own figures (that IS the expand).
            out.forEach(function (e) {
                if (!e._members || !e._members.length) { return; }
                var total = {
                    count: Number(e.count) || 0,
                    metal: Number(e.metal) || 0,
                    crystal: Number(e.crystal) || 0,
                    deuterium: Number(e.deuterium) || 0,
                    units: []
                };
                var seen = {};
                function addUnits(list) {
                    (list || []).forEach(function (u) {
                        var at = seen[u.name];
                        if (at === undefined) { at = seen[u.name] = total.units.length; total.units.push({ name: u.name, amount: 0 }); }
                        total.units[at].amount += Number(u.amount) || 0;
                    });
                }
                addUnits(e.units);
                e._members.forEach(function (m) {
                    total.count += Number(m.count) || 0;
                    total.metal += Number(m.metal) || 0;
                    total.crystal += Number(m.crystal) || 0;
                    total.deuterium += Number(m.deuterium) || 0;
                    addUnits(m.units);
                });
                e._group = total;
            });
            return out;

        function rowsHtml() {

                if (e._members && e._members.length) { html += rowHtml(e, e.mission_id); }
                (e._members || []).forEach(function (m) { html += rowHtml(m, e.mission_id); });
            });

        // Filter bar scaffold: built once per container so an open <select> is never
        // clobbered by a poll rebuild — only the rows div is re-rendered.
        function ensureScaffold(container) {
            syncTypeFilter();
            if (!container._evRows) {
                // A container may point the filter at an external slot (the bottom
                // bubble puts it in its card header so it costs no list row); the top
                // panel keeps an in-list filter bar.
                var slotId = container.getAttribute('data-filter-slot');
                var slot = slotId ? document.getElementById(slotId) : null;
                var filterHtml = '<label>Show: <select class="ev-filter"></select></label>';
                if (slot) {
                    slot.innerHTML = filterHtml;
                    container.innerHTML = '<div class="ev-rowswrap"></div>';
                } else {
                    container.innerHTML = '<div class="ev-filterbar">' + filterHtml + '</div><div class="ev-rowswrap"></div>';
                }
                var sel = (slot || container).querySelector('.ev-filter');
                sel.addEventListener('change', function () {
                    typeFilter = sel.value;
                    try { localStorage.setItem('neoFleetTypeFilter', typeFilter); } catch (err) {}
                    renderPanel();
                });
                sel.addEventListener('click', function (ev) { ev.stopPropagation(); });
                container._evRows = container.querySelector('.ev-rowswrap');
                container._evSel = sel;
            }
            var labels = {};
            events.forEach(function (e) { labels[baseLabel(e)] = 1; });
            var keys = Object.keys(labels).sort();
            var sel = container._evSel;
            var want = ['all'].concat(keys).join('|');

    // ---- Mobile: hide the resource bar on scroll-down, reveal on scroll-up (#108 issue 1). ----
    // The sticky header's 7 resource chips wrap to 2-3 rows on a phone and eat a lot of
    // vertical space. Hide them (and the utility nav) while reading down the page; a small
    // upward swipe brings them back. CSS only acts on body.neo-hidebar inside the mobile
    // breakpoint, so desktop is untouched.
    //
    // ⚠️ Hiding the bar CHANGES THE PAGE HEIGHT (.neo-topwrap is sticky, i.e. still in
    // flow), so the browser re-anchors the scroll and fires another scroll event — with
    // a delta of exactly the header height we just removed. The first version compared
    // against a 6px threshold, which that self-induced delta always exceeds, so every
    // hide immediately looked like a scroll-up and un-hid: the bar strobed for as long
    // as you scrolled, and the constant reflow made tooltips and taps unusable. Three
    // things keep it stable: ignore scroll for a moment after we toggle (the shift is
    // ours, not the user's), require accumulated travel in ONE direction rather than a
    // per-frame delta (a slow drag never crosses a big per-frame threshold), and only
    // run at all on the phone breakpoint the CSS acts on.
    (function () {
        var mq = window.matchMedia('(max-width: 560px)');
        var HIDE_BELOW = 140;   // never hide near the top of the page
        var TRAVEL = 48;        // accumulated px in one direction before toggling
        var SETTLE_MS = 300;    // ignore scroll while our own layout shift settles

        var lastY = window.pageYOffset || 0;
        var travel = 0;
        var settleUntil = 0;
        var ticking = false;

        function setHidden(hide) {
            if (document.body.classList.contains('neo-hidebar') === hide) { return false; }
            document.body.classList.toggle('neo-hidebar', hide);
            return true;
        }

        function update() {
            ticking = false;
            var y = window.pageYOffset || 0;

            // Desktop/tablet: the CSS rule does not apply, so make sure we never leave
            // the class behind after a rotate or resize.
            if (!mq.matches) { setHidden(false); lastY = y; travel = 0; return; }

            if (Date.now() < settleUntil) { lastY = y; travel = 0; return; }

            var delta = y - lastY;
            lastY = y;
            if ((delta > 0) !== (travel > 0)) { travel = 0; }  // direction change starts over
            travel += delta;

            var toggled;
            if (y <= HIDE_BELOW) {
                toggled = setHidden(false);
            } else if (travel > TRAVEL) {
                toggled = setHidden(true);
            } else if (travel < -TRAVEL) {
                toggled = setHidden(false);
            }

            if (toggled) {
                travel = 0;
                settleUntil = Date.now() + SETTLE_MS;
                lastY = window.pageYOffset || 0;
            }
        }

        window.addEventListener('scroll', function () {
            if (!ticking) { window.requestAnimationFrame(update); ticking = true; }
        }, { passive: true });

        if (mq.addEventListener) { mq.addEventListener('change', update); }
    })();

    // ---- Compact-tile hover tooltip (research/shipyard/defense). Each tile carries a
    // hidden .neo-tip-tpl; clone it into a body-appended fixed host on hover (never show

    // native title attribute on tiles that have a rich tooltip.
    (function () {
        // Touch devices synthesize a mouseover on tap, so this hover card would flash
        // open and be closed again by the tap's own click — noise on the one input mode
        // that cannot hover. The tile's detail card is the real UI there.
        if (window.matchMedia('(hover: none) and (pointer: coarse)').matches) { return; }

        var host = null;
        var showTimer = 0;
        function ensureHost() {
            if (!host) {
                host = document.createElement('div');
                host.id = 'neoTip';
                document.body.appendChild(host);
            }
            return host;
        }
        function hideTip() {
            clearTimeout(showTimer);
            if (host) { host.classList.remove('on'); }
        }
        function showTip(tile) {
            var tpl = tile.querySelector('.neo-tip-tpl');
            if (!tpl) { return; }
            if (tile.hasAttribute('title')) { tile.removeAttribute('title'); }
            var h = ensureHost();
            h.innerHTML = tpl.innerHTML;
            h.classList.add('on');
            // Position: below the tile, centered; clamp to viewport; flip above when cramped.
            var r = tile.getBoundingClientRect();
            h.style.left = '0px'; h.style.top = '0px'; // reset before measuring
            var w = h.offsetWidth, hh = h.offsetHeight;
            var x = Math.max(8, Math.min(r.left + r.width / 2 - w / 2, window.innerWidth - w - 8));
            var y = r.bottom + 8;
            if (y + hh > window.innerHeight - 8) { y = r.top - hh - 8; }
            h.style.left = x + 'px';
            h.style.top = Math.max(8, y) + 'px';
        }
        document.addEventListener('mouseover', function (e) {
            var tile = e.target.closest('.neo-tiles-research .neo-tile');
            if (!tile || tile.classList.contains('neo-tile-divider')) { hideTip(); return; }
            if (e.relatedTarget && tile.contains(e.relatedTarget)) { return; } // still inside same tile
            clearTimeout(showTimer);
            showTimer = setTimeout(function () { showTip(tile); }, 140);
        });
        document.addEventListener('mouseout', function (e) {
            var tile = e.target.closest('.neo-tiles-research .neo-tile');
            if (tile && e.relatedTarget && tile.contains(e.relatedTarget)) { return; }
            if (tile) { hideTip(); }
        });
        document.addEventListener('click', hideTip, true);
        window.addEventListener('scroll', hideTip, true);
    })();

    // ---- Click-outside closes the report "More details" overlay. ----
    // The espionage/battle report opens as a jQuery-UI dialog (.overlayDiv carrying
    // .data('uiDialog') and holding a .detail_msg). Its own X (.js_actionKillDetail)
    // does a server round-trip + full page reload; clicking outside should be a light
    // dismiss instead, so we call the dialog's own close() (defaultClose = remove).
    // Backdrop-agnostic: fires whether or not the dialog was opened modal, and a
    // guard keeps a dialog stacked ABOVE the report (e.g. the Send-recyclers fleet
    // popup) from dismissing the report underneath it.
    $(function () {
        $(document).on('mousedown.neoReportOutside', function (e) {
            var $detail = $('.overlayDiv').filter(function () {
                var $d = $(this);
                if (!$d.data('uiDialog') || !$d.find('.detail_msg').length) { return false; }
                try { return $d.dialog('isOpen'); } catch (err) { return false; }
            });
            if (!$detail.length) { return; }
            var $t = $(e.target);
            // Clicks inside any dialog chrome / tooltip / dropdown / confirm box keep it open.
            // .neo-modal-wrap is the full-screen backdrop; clicks land on the wrap, not .neo-modal.
            // .amt-units is the magnitude-shorthand chip strip: like a Tipped tooltip it is
            // body-level by construction (a transformed ancestor would scope its fixed
            // position), so a chip belonging to an input INSIDE the dialog is "outside" it.
            if ($t.closest('.ui-dialog, .ui-tooltip, .tipped, .markItUpDropMenu, #chatBar, .neo-modal, .neo-modal-wrap, .amt-units').length) { return; }
            // Only dismiss when the report is the top-most dialog — never when another
            // dialog (fleet dispatch, confirm) sits above it.
            var maxZ = -1, $top = null;
            $('.ui-dialog:visible').each(function () {
                var z = parseInt($(this).css('zIndex'), 10) || 0;
                if (z >= maxZ) { maxZ = z; $top = $(this); }
            });
            if ($top && !$top.find('.detail_msg').length) { return; }
            $detail.last().dialog('close');
        });

    // ---- Click-outside closes every OTHER bundle overlay dialog (techtree, jump
    // gate, change nick, missile attack, …). ----
    // The bundle already implements this, and neo has never run it: the handler
    // lives in initHideElements(), which is only reached from initIndex() — and neo
    // deliberately does not call initIndex() (it would drag ogameDropDown's
    // select-mangling along). initOverlays() — which neo DOES call, just above —
    // only binds the delegate that OPENS an overlay. So an overlay could only ever
    // be dismissed with its X (#154). Same trap family as `a.build-faster`: the
    // behaviour exists in the bundle and never had its binder run.
    //
    // Modelled on the bundle's own logic, including its two exemptions: a click
    // inside any dialog, and a click inside a tooltip (Tipped renders tooltips
    // OUTSIDE the dialog even when they belong to it, so treating them as "outside"
    // would close the dialog while the user reads its own tooltip).
    //
    // The report overlay keeps its dedicated handler above (extra rules: only the
    // report when it is the global top dialog). This general handler skips report
    // overlays AND bails when any other visible ui-dialog sits above the chosen
    // overlay, so the two handlers never race (one outside click peels one layer).
    $(function () {
        $(document).on('mousedown.neoOverlayOutside', function (e) {
            var $t = $(e.target);
            // .neo-modal-wrap is the full-screen backdrop; clicks land on the wrap, not .neo-modal.
            // .amt-units is the magnitude-shorthand chip strip — same exemption, same reason
            // as the tooltips beside it: it is rendered on <body> and anchored to an input
            // that may live inside this very dialog (reported 2026-09-11: tapping Qa in the
            // resource market closed the trade dialog instead of setting the amount).
            if ($t.closest('.ui-dialog, .ui-tooltip, .tipped, .tpd-tooltip, .markItUpDropMenu, #chatBar, .neo-modal, .neo-modal-wrap, .amt-units').length) {
                return;
            }

            var $open = $('.overlayDiv').filter(function () {
                var $d = $(this);
                if (!$d.data('uiDialog')) { return false; }
                // Handled above, with stricter rules.
                if ($d.find('.detail_msg').length) { return false; }
                try { return $d.dialog('isOpen'); } catch (err) { return false; }
            });
            if (!$open.length) { return; }

            // Close the top-most non-report overlay only, so stacked overlays peel
            // one at a time instead of the whole stack vanishing on one stray click.
            var maxZ = -1, $target = null;
            $open.each(function () {
                var z = parseInt($(this).closest('.ui-dialog').css('zIndex'), 10) || 0;
                if (z >= maxZ) { maxZ = z; $target = $(this); }
            });
            if (!$target) { return; }

            // Also require that overlay to be the global top-most ui-dialog. Without
            // this, a report (or any other dialog) stacked above would still leave
            // this handler free to close the underlay on the same mousedown that
            // the report handler uses to close itself.
            var globalMaxZ = -1, $globalTop = null;
            $('.ui-dialog:visible').each(function () {
                var z = parseInt($(this).css('zIndex'), 10) || 0;
                if (z >= globalMaxZ) { globalMaxZ = z; $globalTop = $(this); }
            });
            if ($globalTop && $globalTop[0] !== $target.closest('.ui-dialog')[0]) { return; }

            $target.dialog('close');
        });
    });
</script>

    // ---- #296 planet reorder: Edit -> drag rows (or A-Z / Coords preset) -> Save. ----
    $(function () {
        var panel = document.getElementById('neoPlanetPanel');
        var editBtn = document.getElementById('neoOrderEdit');
        if (!panel || !editBtn) { return; }
        var bar = document.getElementById('neoOrderBar');
        var editing = false;
        var snapshot = [];
        var dragRow = null;

        var rows = function () {
            return Array.prototype.slice.call(panel.querySelectorAll('.neo-planetwrap'));
        };
        var setEditing = function (on) {
            editing = on;
            panel.classList.toggle('is-ordering', on);
            bar.style.display = on ? '' : 'none';
            editBtn.style.display = on ? 'none' : '';
            rows().forEach(function (row) {
                if (on) { row.setAttribute('draggable', 'true'); } else { row.removeAttribute('draggable'); }
                // Anchors are natively draggable: without this the browser
                // drags the LINK (ghost = its URL) instead of the row.
                row.querySelectorAll('a').forEach(function (a) {
                    if (on) { a.setAttribute('draggable', 'false'); } else { a.removeAttribute('draggable'); }
                });
            });
        };

        editBtn.addEventListener('click', function () {
            snapshot = rows();
            setEditing(true);
        });

        // While editing, planet links must not navigate — a drag that ends on
        // the row it started from would otherwise switch planets.
        panel.addEventListener('click', function (e) {
            if (editing && e.target.closest('a')) { e.preventDefault(); e.stopPropagation(); }
        }, true);

        panel.addEventListener('dragstart', function (e) {
            var row = e.target.closest('.neo-planetwrap');
            if (!editing || !row) { return; }
            dragRow = row;
            row.classList.add('dragging');
            e.dataTransfer.effectAllowed = 'move';
            try {
                e.dataTransfer.setData('text/plain', row.getAttribute('data-planet-id'));
                var box = row.getBoundingClientRect();
                e.dataTransfer.setDragImage(row, e.clientX - box.left, e.clientY - box.top);
            } catch (err) {}
        });
        panel.addEventListener('dragend', function () {
            if (dragRow) { dragRow.classList.remove('dragging'); dragRow = null; }
        });
        panel.addEventListener('dragover', function (e) {
            if (!editing || !dragRow) { return; }
            e.preventDefault();
            e.dataTransfer.dropEffect = 'move';
            // First non-dragged row whose midpoint sits below the cursor: the
            // dragged row slots in front of it (or at the end when past all).
            var next = null;
            var list = rows();
            for (var i = 0; i < list.length; i++) {
                if (list[i] === dragRow) { continue; }
                var box = list[i].getBoundingClientRect();
                if (e.clientY < box.top + box.height / 2) { next = list[i]; break; }
            }
            if (next === dragRow.nextElementSibling) { return; }
            if (next === null) { panel.appendChild(dragRow); } else { panel.insertBefore(dragRow, next); }
        });
        panel.addEventListener('drop', function (e) { if (editing) { e.preventDefault(); } });

        var applyPreset = function (compare) {
            rows().sort(compare).forEach(function (row) { panel.appendChild(row); });
        };
        document.getElementById('neoOrderAz').addEventListener('click', function () {
            applyPreset(function (a, b) {
                return a.getAttribute('data-pname').localeCompare(b.getAttribute('data-pname'))
                    || Number(a.getAttribute('data-planet-id')) - Number(b.getAttribute('data-planet-id'));
            });
        });
        document.getElementById('neoOrderCoords').addEventListener('click', function () {
            applyPreset(function (a, b) {
                var ca = a.getAttribute('data-pcoord').split(',').map(Number);
                var cb = b.getAttribute('data-pcoord').split(',').map(Number);
                return (ca[0] - cb[0]) || (ca[1] - cb[1]) || (ca[2] - cb[2]);
            });
        });
        document.getElementById('neoOrderCancel').addEventListener('click', function () {
            snapshot.forEach(function (row) { panel.appendChild(row); });
            setEditing(false);
        });
        document.getElementById('neoOrderSave').addEventListener('click', function () {
            var order = rows().map(function (row) { return row.getAttribute('data-planet-id'); });
            var btn = this;
            btn.disabled = true;
            $.post("https:\/\/antares.deuteria.net\/ajax\/planet-order", { order: order, _token: '4iNdVFA37xfvkVoElHbQz1LFpVLgkbBaC8oGrNge' }, function (res) {
                btn.disabled = false;
                if (res && res.success) { setEditing(false); return; }
                fadeBox((res && res.error) || "No se pudo guardar el orden de los planetas.", true);
            }, 'json').fail(function (xhr) {
                btn.disabled = false;
                var msg = (xhr.responseJSON && xhr.responseJSON.error) || "No se pudo guardar el orden de los planetas.";
                fadeBox(msg, true);
            });
        });
    });
</script>
